@Composable
fun WeatherApp() {
    var country by remember { mutableStateOf("Colombia") } // Changed default for testing
    var weatherData by remember { mutableStateOf("Enter a country and press 'Get Weather'.") }
    val scope = rememberCoroutineScope()

    Surface(
        modifier = Modifier.fillMaxSize(),
        color = MaterialTheme.colorScheme.background
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(16.dp)
        ) {
            TextField(
                value = country,
                onValueChange = { country = it },
                label = { Text("Enter Country o Pais") },
                modifier = Modifier.fillMaxWidth(),
                singleLine = true
            )

            Spacer(Modifier.height(8.dp))

            Button(
                onClick = {
                    weatherData = "Loading weather data for $country..."
                    scope.launch {
                        weatherData = fetchWeatherData(country)
                    }
                },
                modifier = Modifier.fillMaxWidth()
            ) {
                Text("Get Weather o Ver Clima")
            }
            Spacer(Modifier.height(16.dp))
            val scrollState = rememberScrollState()
            Column(modifier = Modifier.verticalScroll(scrollState)) {
                Text(text = weatherData)
            }
        }
    }
}
fun formatWeatherData(response: WeatherResponse, timezoneId: String?): String {
    val values = response.data.values
    val locationName = response.location.name
    val timestamp = response.data.time

    val utcDateTime = ZonedDateTime.parse(timestamp)
    var timezoneNote = ""

    val targetZoneId = try {
        if (timezoneId != null) {
            ZoneId.of(timezoneId)
        } else {
            timezoneNote = "\n\n(Note: Could not fetch timezone. Displaying UTC time.)"
            utcDateTime.zone 
        }
    } catch (e: Exception) {
        timezoneNote = "\n\n(Note: Invalid timezone ID '$timezoneId'. Displaying UTC time.)"
        utcDateTime.zone 
    }

    val localDateTime = utcDateTime.withZoneSameInstant(targetZoneId)

    val englishFormatter = DateTimeFormatter.ofPattern("MMMM d, yyyy - h:mm a", Locale.ENGLISH)
    val spanishFormatter = DateTimeFormatter.ofPattern("d 'de' MMMM 'de' yyyy - h:mm a", Locale("es", "ES"))
    val formattedDateEnglish = localDateTime.format(englishFormatter)
    val formattedDateSpanish = localDateTime.format(spanishFormatter)

    val english = """
        --- Weather in English ---
        Location: $locationName
        Date & Time: $formattedDateEnglish
        Temperature: ${values.temperature}°C
        Wind Speed: ${values.windSpeed} kph
        Wind Direction: ${values.windDirection}°
        Precipitation Probability: ${values.precipitationProbability}%
        Pressure: ${values.pressureSurfaceLevel} hPa
        Rain Intensity: ${values.rainIntensity}
    """.trimIndent()

    val spanish = """
        --- Clima en Español ---
        Ubicación: $locationName
        Fecha y Hora: $formattedDateSpanish
        Temperatura: ${values.temperature}°C
        Velocidad del Viento: ${values.windSpeed} kph
        Dirección del Viento: ${values.windDirection}°
        Probabilidad de Precipitación: ${values.precipitationProbability}%
        Presión: ${values.pressureSurfaceLevel} hPa
        Intensidad de Lluvia: ${values.rainIntensity}
    """.trimIndent()

    return "$english\n\n$spanish$timezoneNote"
}
suspend fun fetchWeatherData(country: String): String = withContext(Dispatchers.IO) {
    val client = OkHttpClient()
    val gson = Gson()

    val encodedLocation = URLEncoder.encode(country, "UTF-8")
    val apiKey = "TNOdFaS3CHMJCLaEaF211QR8cURuGJqW"
    val weatherRequest = Request.Builder()
        .url("https://api.tomorrow.io/v4/weather/realtime?location=$encodedLocation&apikey=$apiKey")
        .get()
        .addHeader("accept", "application/json")
        .build()

    try {
        val weatherApiResponse = client.newCall(weatherRequest).execute()
        if (weatherApiResponse.isSuccessful) {
            val weatherBody = weatherApiResponse.body?.string()
            if (weatherBody != null) {
                val weatherResponse = gson.fromJson(weatherBody, WeatherResponse::class.java)

                val lat = weatherResponse.location.lat
                val lon = weatherResponse.location.lon
                val tzRequest = Request.Builder()
                    .url("https://timeapi.io/api/TimeZone/coordinate?latitude=$lat&longitude=$lon")
                    .get().build()

                try {
                    val tzApiResponse = client.newCall(tzRequest).execute()
                    if (tzApiResponse.isSuccessful) {
                        val tzBody = tzApiResponse.body?.string()
                        if (tzBody != null) {
                            val timeZoneResponse = gson.fromJson(tzBody, TimeZoneResponse::class.java)
                            return@withContext formatWeatherData(weatherResponse, timeZoneResponse.timeZone)
                        }
                    }
                } catch (e: Exception) {
                 
                }

                return@withContext formatWeatherData(weatherResponse, null)

            } else {
                return@withContext "Error: Empty response from server."
            }
        } else {
            return@withContext "Error fetching data: ${weatherApiResponse.code} ${weatherApiResponse.message}"
        }
    } catch (e: Exception) { 
        return@withContext when (e) {
            is java.net.UnknownHostException -> "Error: Unable to connect. Please check your internet connection."
            else -> "Error: ${e.message}"
        }
    }
}
